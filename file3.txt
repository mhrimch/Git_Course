import { IcInfoCircleS } from '@finanzinformatik/mcd-module-icons';
import { useMcdPalette } from '@finanzinformatik/mcd-module-theme-provider';
import {
  Box,
  Button,
  Card,
  CardActions,
  CardContent,
  ClickAwayListener,
  Icon,
  Skeleton,
  Stack,
  SvgIconProps,
  Tooltip,
  Typography,
  useMediaQuery,
} from '@mui/material';
import { useTheme } from '@mui/system';
import { ThemeConstants } from '@personalsysteme/core';
import { useState } from 'react';

interface QuickActionInfoCardProps {
  HeaderIcon: (props: SvgIconProps) => React.ReactElement;
  title?: string;
  subtitle?: string;
  buttonLabel?: string;
  onClick?: () => void;
  aktionVerfuegbar: boolean;
  hasInfoIcon: boolean;
}

export const QuickActionInfoCard = ({
  HeaderIcon,
  title,
  subtitle,
  buttonLabel,
  onClick,
  aktionVerfuegbar,
  hasInfoIcon,
}: QuickActionInfoCardProps) => {
  const theme = useTheme();
  const isDesktop = useMediaQuery(theme.breakpoints.up('sm'));
  const { ColorTokens } = useMcdPalette();
  const [openTooltip, setOpenTooltip] = useState<boolean>(false);
  return (
    <Card
      sx={{
        border: isDesktop ? 0 : 1,
        borderColor: 'divider',
        boxShadow: 'none',
      }}
    >
      <Box display={'flex'} justifyContent={'center'}>
        <HeaderIcon color={'primary'} sx={{ fontSize: ThemeConstants.ICON_SIZE_LG }} />
      </Box>
      <CardContent sx={{ flexGrow: 1, padding: 0 }}>
        <Stack display={'flex'} direction="column" alignItems="center">
          <Typography variant={ThemeConstants.TYPOGRAPHY_HEADLINE_02}>{title}</Typography>
          <Stack display={'flex'} direction="row">
            <Typography
              variant={ThemeConstants.TYPOGRAPHY_MULTILINE_02}
              color={ColorTokens[ThemeConstants.COLOR_TOKEN_READABLE_02]}
            >
              {subtitle}
            </Typography>
            <ClickAwayListener onClickAway={() => setOpenTooltip(false)}>
              <Tooltip title={'Tooltip information'} open={openTooltip}>
                {hasInfoIcon && (
                  <Icon
                    sx={{
                      display: 'flex',
                      alignItems: 'center',
                      ml: 0.5,
                      height: '1.2rem',
                      width: '1.2rem',
                      cursor: 'pointer',
                    }}
                  >
                    <IcInfoCircleS fontSize={'small'} />
                  </Icon>
                )}
              </Tooltip>
            </ClickAwayListener>
          </Stack>
        </Stack>
      </CardContent>
      <CardActions>
        {aktionVerfuegbar && (
          <Button onClick={onClick} variant={'outlined'}>
            {buttonLabel}
          </Button>
        )}
      </CardActions>
    </Card>
  );
};
export const QuickActionInfoCardSkeleton = () => {
  const theme = useTheme();
  const isDesktop = useMediaQuery(theme.breakpoints.up('sm'));
  return (
    <Card
      sx={{
        border: isDesktop ? 0 : 1,
        borderColor: 'divider',
        boxShadow: 'none',
      }}
    >
      <Box display={'flex'} justifyContent={'center'}>
        <Skeleton animation={ThemeConstants.ANIMATION_WAVE} variant={ThemeConstants.VARIANT_CIRCULAR}></Skeleton>
      </Box>
      <CardContent>
        <Stack display={'flex'} direction="column" alignItems="center">
          <Skeleton animation={ThemeConstants.ANIMATION_WAVE}></Skeleton>
          <Skeleton animation={ThemeConstants.ANIMATION_WAVE}></Skeleton>
        </Stack>
      </CardContent>
      <CardActions>
        <Skeleton>
          <Button />
        </Skeleton>
      </CardActions>
    </Card>
  );
};
