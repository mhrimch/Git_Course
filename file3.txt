import { IcInfoCircleS } from '@finanzinformatik/mcd-module-icons';
import { useMcdPalette } from '@finanzinformatik/mcd-module-theme-provider';
import {
  Box,
  Button,
  Card,
  CardActions,
  CardContent,
  Icon,
  Skeleton,
  Stack,
  SvgIconProps,
  Tooltip,
  Typography,
  useMediaQuery,
} from '@mui/material';
import { useTheme } from '@mui/system';
import React from 'react';
import { ThemeConstants } from '@personalsysteme/core';

interface QuickActionInfoCardProps {
  HeaderIcon: (props: SvgIconProps) => React.ReactElement;
  title?: string;
  subtitle?: string;
  buttonLabel?: string;
  onClick?: () => void;
  aktionVerfuegbar: boolean;
}

export const QuickActionInfoCard = ({
  HeaderIcon,
  title,
  subtitle,
  buttonLabel,
  onClick,
  aktionVerfuegbar,
}: QuickActionInfoCardProps) => {
  const theme = useTheme();
  const isDesktop = useMediaQuery(theme.breakpoints.up('sm'));
  const { ColorTokens } = useMcdPalette();

  const tooltipContent = (
    <Box>
      <Typography variant={ThemeConstants.TYPOGRAPHY_MULTILINE_02}>
        Die verfügbaren Urlaubstage beinhalten noch nicht verplante Resturlaubstage.
      </Typography>
      <Typography variant={ThemeConstants.TYPOGRAPHY_MULTILINE_02} color={'error'}>
        Bereits beantragte, aber noch nicht genehmigte Urlaubsanträge sind berücksichtigt.
      </Typography>
    </Box>
  );

  return (
    <Card
      sx={{
        border: isDesktop ? 0 : 1,
        borderColor: 'divider',
        boxShadow: 'none',
        flex: 1,
      }}
    >
      <Box display={'flex'} justifyContent={'center'}>
        <HeaderIcon color={'primary'} sx={{ fontSize: ThemeConstants.ICON_SIZE_LG }} />
      </Box>
      <CardContent sx={{ flexGrow: 1 }}>
        <Stack direction="column" alignItems="center">
          <Typography variant={ThemeConstants.TYPOGRAPHY_HEADLINE_02}>{title}</Typography>
          <Stack direction="row" alignItems="center">
            <Typography
              variant={ThemeConstants.TYPOGRAPHY_MULTILINE_02}
              color={ColorTokens[ThemeConstants.COLOR_TOKEN_READABLE_02]}
            >
              {subtitle}
            </Typography>
            <Tooltip title={tooltipContent} arrow>
              <Icon sx={{ display: 'flex', alignItems: 'center', ml: 0.5, cursor: 'pointer' }}>
                <IcInfoCircleS fontSize={'small'} />
              </Icon>
            </Tooltip>
          </Stack>
        </Stack>
      </CardContent>
      <CardActions sx={{ justifyContent: 'center' }}>
        {aktionVerfuegbar && (
          <Button onClick={onClick} variant={'outlined'}>
            {buttonLabel}
          </Button>
        )}
      </CardActions>
    </Card>
  );
};

export const QuickActionInfoCardSkeleton = () => {
  const theme = useTheme();
  const isDesktop = useMediaQuery(theme.breakpoints.up('sm'));
  return (
    <Card
      sx={{
        border: isDesktop ? 0 : 1,
        borderColor: 'divider',
        boxShadow: 'none',
        flex: 1,
      }}
    >
      <Box display={'flex'} justifyContent={'center'}>
        <Skeleton animation={ThemeConstants.ANIMATION_WAVE} variant={ThemeConstants.VARIANT_CIRCULAR} />
      </Box>
      <CardContent>
        <Stack direction="column" alignItems="center">
          <Skeleton animation={ThemeConstants.ANIMATION_WAVE} />
          <Skeleton animation={ThemeConstants.ANIMATION_WAVE} />
        </Stack>
      </CardContent>
      <CardActions>
        <Skeleton>
          <Button />
        </Skeleton>
      </CardActions>
    </Card>
  );
};
