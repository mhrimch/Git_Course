const handleAnzahlTageChange = (e: ChangeEvent<HTMLInputElement>) => {
  const raw = e.target.value;

  if (raw === '') {
    setErrors((prev) => ({ ...prev, anzahlTage: null }));
    setKonfAntrag((prev) => ({ ...prev, anzahlTageRueckwirkenderKorrektur: '' }));
    return;
  }

  const parsed = parseInt(raw, 10);

  if (!Number.isInteger(parsed) || parsed < 0 || parsed > 180) {
    setErrors((prev) => ({ ...prev, anzahlTage: 'Ganztägige Werte zwischen 0 und 180 möglich' }));
    return;
  }

  setErrors((prev) => ({ ...prev, anzahlTage: null }));
  setKonfAntrag((prev) => ({ ...prev, anzahlTageRueckwirkenderKorrektur: parsed }));
};
