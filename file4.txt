interface AntraegeKonfigCheckboxPanelProps {
  versionsnummer: string;
  buchungskreis: string;
  antragsgruppe: KonfAntragsgruppeRead;
  antrag: KonfAntragRead;
  antragAktiv: boolean;
  isAntragInPersonalservicesAktiv?: boolean;
  dokumententype: Dokumententyp[];
}

export const KonfAntragEigenschaftenPanel = ({
  versionsnummer,
  buchungskreis,
  antrag,
  antragsgruppe,
  antragAktiv,
  isAntragInPersonalservicesAktiv,
  dokumententype,
}: AntraegeKonfigCheckboxPanelProps) => {
  const { t } = useTranslation();
  const queryClient = useQueryClient();
  const [konfAntrag, setKonfAntrag] = useState<KonfAntragRead>(antrag);
  const [urlTouched, setUrlTouched] = useState(false);
  const [urlError, setUrlError] = useState(false);
  const [anzahltageError, setAnzahlTageError] = useState(false);
  const [uhrzeitChecked, setUhrzeitChecked] = useState(true);
  const { mutateAsync: updateKonfAntrag } = useUpdateKonfAntraege();
  const workflowService = useWorkflowService();
  const notificationStoreActions = useNotificationStoreActions();
  const isZeitkorrektur = antragsgruppe.id === AntragsgruppenIds.zeitkorrektur;
  const updateEigenschaft = (event: ChangeEvent<HTMLInputElement>) => {
    setKonfAntrag((prev) => ({ ...prev, [event.target.name]: event.target.checked }));
  };

  const handleGanztaegigChange = (event: ChangeEvent<HTMLInputElement>) => {
    if (!konfAntrag.isUhrzeit && konfAntrag.isGanztaegig) {
      notificationStoreActions.addInfoNotification(
        'Aktivieren Sie mindestens eine Aktion "Uhrzeit zulassen" oder "ganztägig zulassen'
      );
      return;
    }

    setKonfAntrag((prev) => ({ ...prev, isGanztaegig: event.target.checked }));
  };

  const handleUhrzeitChange = (event: ChangeEvent<HTMLInputElement>) => {
    if (konfAntrag.isUhrzeit && !konfAntrag.isGanztaegig) {
      notificationStoreActions.addInfoNotification(
        'Aktivieren Sie mindestens eine Aktion "Uhrzeit zulassen" oder "ganztägig zulassen'
      );
      return;
    }

    setUhrzeitChecked(event.target.checked);
    setKonfAntrag((prev) => ({ ...prev, isUhrzeit: event.target.checked }));
  };

  const handleAblageOrtSelect = (event: SelectChangeEvent) => {
    setKonfAntrag((prev) => ({
      ...prev,
      epaArt: event.target.value,
    }));
  };

  const handleAnzahlTageChange = (event: ChangeEvent<HTMLInputElement>) => {
    const raw = event.target.value;
    if (raw === '') {
      setKonfAntrag((prev) => ({
        ...prev,
        anzahlTageRueckwirkenderKorrektur: 0,
      }));
      return;
    }

    const parsed = parseInt(raw, 10);
    if (!Number.isInteger(parsed) || parsed < 0 || parsed > 180) {
      setAnzahlTageError(true);
      return;
    }
    setAnzahlTageError(false);
    setKonfAntrag((prev) => ({ ...prev, anzahlTageRueckwirkenderKorrektur: parsed }));
  };

  const handleUrlChange = (event: ChangeEvent<HTMLInputElement>) => {
    const value = event.target.value;
    setKonfAntrag((prev) => ({ ...prev, url: value }));
  };
  const handleUrlBlur = () => {
    setUrlTouched(true);
    if (konfAntrag.url.trim() === '') {
      setUrlError(false);
      return;
    }
    if (isValidUrl(konfAntrag.url)) {
      setUrlError(false);
    } else {
      setUrlError(true);
    }
  };

  const isValidUrl = (value: string): boolean => {
    try {
      const normalisierteURL = value.startsWith('http') || value.startsWith('https') ? value : `https://${value}`;
      const url = new URL(normalisierteURL);
      const isHttp = url.protocol === 'http:' || url.protocol === 'https:';
      const hasValidHost = url.hostname.includes('.');
      return isHttp && hasValidHost;
    } catch {
      return false;
    }
  };

  const handleClickSpeichern = () => {
    if (antrag.isUrl && !isAntragInPersonalservicesAktiv) {
      return;
    }
    const konfAntragWrite: KonfAntragWrite = {
      versionsnummer: versionsnummer,
      buchungskreis: buchungskreis,
      gruppenIconId: antragsgruppe.iconId,
      antragId: konfAntrag.id,
      hinweis: konfAntrag.hinweis,
      isAktiv: antragAktiv,
      isFavorit: konfAntrag.isFavorit,
      isFuehrungskraftFreigabePflicht: konfAntrag.isFuehrungskraftFreigabePflicht,
      isPersonalFreigabePflicht: konfAntrag.isPersonalFreigabePflicht,
      isKommentarPflicht: konfAntrag.isKommentarPflicht,
      isGanztaegig: konfAntrag.isGanztaegig,
      isUhrzeit: konfAntrag.isUhrzeit,
      isSerienelement: konfAntrag.isSerienelement,
      isDocumentUpload: konfAntrag.isDocumentUpload,
      isDocumentUploadPflicht: konfAntrag.isDocumentUploadPflicht,
      epaArt: konfAntrag.epaArt,
      maxAnzahlTage: konfAntrag.maxAnzahlTage,
      maxAnzahlTageEinheit: konfAntrag.maxAnzahlTageEinheit,
      anzahlTageRueckwirkenderKorrektur: konfAntrag.anzahlTageRueckwirkenderKorrektur,
      url: konfAntrag.url,
      isUrl: konfAntrag.isUrl,
      verarbeitungsart: konfAntrag.verarbeitungsart,
      personalserviceSchluessel: konfAntrag.personalserviceSchluessel,
      personalserviceBezeichnung: konfAntrag.personalserviceBezeichnung,
      personalserviceKommentar: konfAntrag.personalserviceKommentar,
    };
    updateKonfAntrag(konfAntragWrite).then(() => {
      queryClient.invalidateQueries({ queryKey: KonfigKeys.konfigAntraege }).then(() => {
        navigateToPskKonfiguration(workflowService);
      });
    });
  };

  return (
    <TableContainer>
      <TableHead>
        <TableRow>
          <TableCell>
            <Typography noWrap={true}>{t('Einstellungen') + ' ' + antrag.bezeichnung}</Typography>
          </TableCell>
          <TableCell width={ThemeConstants.WIDTH_100_PERCENT} />
        </TableRow>
      </TableHead>
      <TableBody>
        <AntragseigenschaftCheckbox
          key={'isFuehrungskraftFreigabePflicht'}
          beschreibung={t('GenehmigungFuehrungskraftErforderlich')}
          status={konfAntrag.isFuehrungskraftFreigabePflicht}
          name={'isFuehrungskraftFreigabePflicht'}
          onChange={updateEigenschaft}
          disabled={false}
        />
        <AntragseigenschaftCheckbox
          key={'isPersonalFreigabePflicht'}
          beschreibung={t('ZentraleGenehmigungErforderlich')}
          status={konfAntrag.isPersonalFreigabePflicht}
          name={'isPersonalFreigabePflicht'}
          onChange={updateEigenschaft}
          disabled={false}
        />

        {!isZeitkorrektur && (
          <AntragseigenschaftCheckbox
            key={'isKommentarPflicht'}
            beschreibung={t('KommentarAlsPflichtfeld')}
            status={konfAntrag.isFuehrungskraftFreigabePflicht ? konfAntrag.isKommentarPflicht : false}
            name={'isKommentarPflicht'}
            onChange={updateEigenschaft}
            disabled={konfAntrag.isUrl}
          />
        )}

        {!isZeitkorrektur && (
          <AntragseigenschaftCheckbox
            key={'isUhrzeit'}
            beschreibung={t('UhrzeitZulassen')}
            status={konfAntrag.isUhrzeit}
            name={'isUhrzeit'}
            onChange={handleUhrzeitChange}
            disabled={false}
          />
        )}

        {!isZeitkorrektur && uhrzeitChecked && (
          <AntragseigenschaftCheckbox
            key={'isGanztaegig'}
            beschreibung={t('ganztaegigZulassen')}
            status={konfAntrag.isGanztaegig}
            name={'isGanztaegig'}
            onChange={handleGanztaegigChange}
            disabled={false}
          />
        )}

        {!isZeitkorrektur && (
          <AntragseigenschaftCheckbox
            key={'isDocumentUpload'}
            beschreibung={t('DokumentenUploadMoeglich')}
            status={konfAntrag.isDocumentUpload}
            name={'isDocumentUpload'}
            onChange={updateEigenschaft}
            disabled={false}
          />
        )}
        {konfAntrag.isDocumentUpload && (
          <Box sx={{ pl: 6 }}>
            <AntragseigenschaftCheckbox
              key={'isDocumentUploadPflicht'}
              beschreibung={t('DokumentenUploadPflicht')}
              status={konfAntrag.isDocumentUploadPflicht}
              name={'isDocumentUploadPflicht'}
              onChange={updateEigenschaft}
              disabled={false}
            />
          </Box>
        )}

        <AntragseigenschaftSelect
          beschreibung={t('AblageortEpa')}
          label={'Dokumenententyp'}
          value={konfAntrag.epaArt}
          name={'epaArt'}
          onChange={handleAblageOrtSelect}
          dokumententype={dokumententype}
        ></AntragseigenschaftSelect>

        <AntragseigenschaftTextfield
          beschreibung={t('AnzahlTageRueckwirkenderKorrektur')}
          label={t('AnzahlTageRueckwirkenderKorrektur')}
          name={'anzahlTageRueckwirkenderKorrektur'}
          value={konfAntrag.anzahlTageRueckwirkenderKorrektur}
          type={'text'}
          helperText={anzahltageError ? t('ZugelasseneGenztaegigeWerte') : t('Max180Tage')}
          error={anzahltageError}
          onChange={handleAnzahlTageChange}
          slotProps={{
            input: {
              error: anzahltageError,
            },
            htmlInput: { min: 0, max: 180 },
          }}
        ></AntragseigenschaftTextfield>

        <AntragseigenschaftTextfield
          beschreibung={t('Intranetlink')}
          label={t('Intranetlink')}
          name={'url'}
          type={'text'}
          value={konfAntrag.url}
          placeholder={t('UrlEingeben')}
          helperText={urlTouched && urlError ? t('UngueltigesFormat') : undefined}
          onChange={handleUrlChange}
          slotProps={{
            input: {
              onBlur: handleUrlBlur,
              endAdornment: (
                <InputAdornment position="end">
                  <Button
                    onClick={() => {
                      if (isValidUrl(konfAntrag.url)) {
                        window.open(konfAntrag.url, '_blank');
                      }
                    }}
                  >
                    {t('linkOeffnen')}
                  </Button>
                </InputAdornment>
              ),
            },
          }}
        />

        <TableRow>
          <TableCell>
            <Button variant="contained" size="large" onClick={handleClickSpeichern}>
              {t('Speichern')}
            </Button>
          </TableCell>
          <TableCell width={ThemeConstants.WIDTH_100_PERCENT} />
        </TableRow>
      </TableBody>
    </TableContainer>
  );
};
